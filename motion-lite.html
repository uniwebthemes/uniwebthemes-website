// ─── motion-lite.js · ~6.2kb minified · 0 dependencies ─── const MotionLite = (() => { // Easing presets (CSS cubic-bezier or keywords) const ease = { spring: 'cubic-bezier(0.34, 1.56, 0.64, 1)', smooth: 'cubic-bezier(0.25, 1, 0.5, 1)', snappy: 'cubic-bezier(0.2, 0, 0, 1)', bounce: 'cubic-bezier(0.34, 1.8, 0.64, 1)', linear: 'linear', }; // Core: animate() wraps Web Animations API function animate(el, keyframes, opts = {}) { const defaults = { duration: 400, easing: ease.smooth, fill: 'forwards', delay: 0, }; const anim = el.animate(keyframes, { ...defaults, ...opts }); return anim; // Gives you .pause() .play() .cancel() } // Stagger: animate multiple elements with delay cascade function stagger(els, keyframes, opts = {}) { const { staggerMs = 60, ...rest } = opts; return [...els].map((el, i) => animate(el, keyframes, { delay: i * staggerMs, ...rest }) ); } // Scroll trigger: fire callback when element enters viewport function scroll(els, onEnter, { threshold = 0.15, once = true } = {}) { const observer = new IntersectionObserver((entries) => { entries.forEach(e => { if (e.isIntersecting) { onEnter(e.target, e); if (once) observer.unobserve(e.target); } }); }, { threshold }); [...els].forEach(el => observer.observe(el)); return observer; } // Timeline: sequence animations with delays function timeline(sequence) { let offset = 0; sequence.forEach(([el, keyframes, opts]) => { animate(el, keyframes, { ...opts, delay: (opts?.delay ?? 0) + offset }); offset += (opts?.duration ?? 400) + (opts?.gap ?? 0); }); } return { animate, stagger, scroll, timeline, ease }; })();
