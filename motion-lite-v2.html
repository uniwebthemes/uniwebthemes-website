Demo 01 — Parallax

Multi-layer depth

Move your mouse (or scroll on mobile) — each layer moves at a different speed, creating real depth. Pure CSS transforms + JS, no canvas, no WebGL.

Mouse Parallax

Move cursor over scene · 6 independent layers

// parallax() — mouse + scroll driven depth layers function parallax(container, { mode = 'mouse' } = {}) { const layers = container.querySelectorAll('[data-parallax-speed]'); if (mode === 'mouse') { container.addEventListener('mousemove', e => { const r = container.getBoundingClientRect(); const cx = (e.clientX - r.left - r.width / 2); const cy = (e.clientY - r.top - r.height / 2); layers.forEach(layer => { const s = parseFloat(layer.dataset.parallaxSpeed); layer.style.transform = `translate(${cx*s}px, ${cy*s}px)`; }); }); } if (mode === 'scroll') { window.addEventListener('scroll', () => { const scrolled = window.scrollY; layers.forEach(layer => { const s = parseFloat(layer.dataset.parallaxSpeed); layer.style.transform = `translateY(${scrolled * s}px)`; }); }, { passive: true }); } }

// createSlider() — blaze-inspired, touch+pointer+wheel+keyboard function createSlider(container) { const track = container.querySelector('.slider-track'); const slides = [...track.querySelectorAll('.slide')]; let current = 0, startX = 0, dragging = false, dragDelta = 0; // Read CSS variables — layout is fully CSS-controlled const getSlideWidth = () => slides[0].getBoundingClientRect().width; const gap = parseFloat(getComputedStyle(container).getPropertyValue('--slide-gap')); const goTo = (index, animated = true) => { current = Math.max(0, Math.min(slides.length - 1, index)); const offset = -current * (getSlideWidth() + gap); track.style.transform = `translateX(${offset + dragDelta}px)`; /* dots, arrows update here */ }; // Pointer drag (covers touch + mouse) track.addEventListener('pointerdown', e => { dragging = true; startX = e.clientX; track.setPointerCapture(e.pointerId); track.classList.remove('is-animating'); }); track.addEventListener('pointermove', e => { if (!dragging) return; dragDelta = e.clientX - startX; goTo(current, false); }); track.addEventListener('pointerup', e => { dragging = false; const threshold = getSlideWidth() * 0.2; if (dragDelta < -threshold) goTo(current + 1); else if (dragDelta > threshold) goTo(current - 1); else goTo(current); dragDelta = 0; }); }
